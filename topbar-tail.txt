                  </Link>
                </DropdownMenuItem>
              </DropdownMenuContent>
            </DropdownMenu>
          </div>
        </div>

        {!isHome && <div className="mx-auto flex max-w-[440px] items-center justify-between gap-3 px-4 pb-3 md:hidden">
          <button
            type="button"
            onClick={() => setSearchOpen(true)}
            className="flex min-w-0 flex-1 items-center gap-2 rounded-[1.05rem] border border-white/14 bg-[linear-gradient(180deg,rgba(17,30,50,0.98),rgba(11,22,39,0.98))] px-3.5 py-3 text-left text-sm font-medium text-white shadow-[0_10px_24px_rgba(0,0,0,0.16)] backdrop-blur"
          >
            <Search className="size-3.5 shrink-0 text-white/75" />
            <span className="truncate text-white/78">Search teams, matches, stories…</span>
          </button>
          {isInstallable && (
            <button
              type="button"
              onClick={promptInstall}
              className="flex shrink-0 items-center gap-1 rounded-[1rem] border border-sky-300/20 bg-sky-400/15 p-3 text-xs font-semibold text-sky-100"
            >
              <Download className="size-3.5" />
              Install
            </button>
          )}
        </div>}
        )}
      </header>
    </>
  );
}
